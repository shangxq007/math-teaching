\begin{multicols}{3}

\begin{fastcard}{集合与逻辑}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $A\cap B$，$A\cup B$，$\complement_U A$
\item $A\subseteq B \iff p\Rightarrow q$
\item 德摩根：$\complement_U(A\cap B)=(\complement_U A)\cup(\complement_U B)$
\end{itemize}
\end{fastcard}

\begin{fastcard}{基本不等式}
$a+b\ge 2\sqrt{ab}$（$a,b>0$），等号当且仅当 $a=b$。
\end{fastcard}

\begin{fastcard}{函数}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 定义域、值域、单调性、奇偶性
\item 反函数：$y=f(x)$ 与 $y=f^{-1}(x)$ 关于 $y=x$ 对称
\end{itemize}
\end{fastcard}

\begin{fastcard}{指数与对数}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $a^m a^n=a^{m+n}$，$(a^m)^n=a^{mn}$
\item $\log_a(MN)=\log_a M+\log_a N$
\item $\log_a b=\dfrac{\ln b}{\ln a}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{数列}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 等差：$a_n=a_1+(n-1)d$，$S_n=\dfrac{n(a_1+a_n)}{2}$
\item 等比：$a_n=a_1q^{n-1}$，$S_n=\dfrac{a_1(1-q^n)}{1-q}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{平面向量}
$\vec a\cdot\vec b=|\vec a||\vec b|\cos\theta$；
垂直 $\iff \vec a\cdot\vec b=0$。
\end{fastcard}

\begin{fastcard}{复数}
$z=a+bi$，$|z|=\sqrt{a^2+b^2}$，$i^2=-1$，$i$ 的幂周期为 4。
\end{fastcard}

\begin{fastcard}{三角函数}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $\sin^2x+\cos^2x=1$，$1+\tan^2x=\sec^2x$
\item $\sin2x=2\sin x\cos x$
\item $T=\dfrac{2\pi}{|\omega|}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{概率统计}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $P(A)=\dfrac{m}{n}$
\item $\bar x=\dfrac1n\sum x_i$，$s^2=\dfrac1n\sum(x_i-\bar x)^2$
\item $P(X=k)=\mathrm C_n^k p^k(1-p)^{n-k}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{排列组合与二项式}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $\mathrm A_n^m=\dfrac{n!}{(n-m)!}$，$\mathrm C_n^m=\dfrac{n!}{m!(n-m)!}$
\item $(a+b)^n=\sum\limits_{k=0}^{n}\mathrm C_n^k a^{n-k}b^k$
\end{itemize}
\end{fastcard}

\begin{fastcard}{解析几何}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 点 $(x_0,y_0)$ 到直线 $Ax+By+C=0$：$d=\dfrac{|Ax_0+By_0+C|}{\sqrt{A^2+B^2}}$
\item 椭圆 $e=\dfrac ca$，双曲线 $e=\dfrac ca$，抛物线 $e=1$
\end{itemize}
\end{fastcard}

\begin{fastcard}{立体几何}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 柱 $V=Sh$，锥 $V=\dfrac13Sh$，球 $V=\dfrac43\pi R^3$
\item 线面角 $\sin\theta=\dfrac{|\vec a\cdot\vec n|}{|\vec a||\vec n|}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{导数}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $(x^n)'=nx^{n-1}$，$(\sin x)'=\cos x$，$(\cos x)'=-\sin x$
\item $(e^x)'=e^x$，$(\ln x)'=\dfrac1x$
\item $f'(x)>0\Rightarrow$ 递增
\item $\displaystyle\int x^n\,\mathrm{d}x=\dfrac{x^{n+1}}{n+1}+C$
\end{itemize}
\end{fastcard}

\end{multicols}
