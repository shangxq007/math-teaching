\begin{fastcard}{同角与诱导}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $\sin^2x+\cos^2x=1$，$1+\tan^2x=\sec^2x$
\item 诱导：奇变偶不变，符号看象限
\end{itemize}
\end{fastcard}

\begin{fastcard}{三角恒等}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $\sin2x=2\sin x\cos x$
\item $\cos2x=\cos^2x-\sin^2x$
\item $a\sin x+b\cos x=\sqrt{a^2+b^2}\sin(x+\varphi)$
\end{itemize}
\end{fastcard}

\begin{fastcard}{$y=A\sin(\omega x+\varphi)$}
$T=\dfrac{2\pi}{|\omega|}$，振幅 $|A|$，初相 $\varphi$。
\end{fastcard}

\begin{fastcard}{解三角形}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 正弦定理：$\dfrac{a}{\sin A}=\dfrac{b}{\sin B}=\dfrac{c}{\sin C}=2R$
\item 余弦定理：$c^2=a^2+b^2-2ab\cos C$
\item 面积：$S=\dfrac12 ab\sin C$
\end{itemize}
\end{fastcard}

\begin{fastcard}{平面向量}
$\vec a\cdot\vec b=|\vec a||\vec b|\cos\theta$；
垂直 $\iff \vec a\cdot\vec b=0$。
\end{fastcard}

\begin{fastcard}{立体几何}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item 柱体 $V=Sh$，锥体 $V=\dfrac13Sh$，球 $V=\dfrac43\pi R^3$
\item 球面 $S=4\pi R^2$
\item 线面角：$\sin\theta=\dfrac{|\vec a\cdot\vec n|}{|\vec a||\vec n|}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{概率统计}
\begin{itemize}[leftmargin=1.2em, topsep=0.2ex, itemsep=0.1ex, parsep=0pt]
\item $P(A)=\dfrac{m}{n}$
\item $\bar x=\dfrac1n\sum x_i$，$s^2=\dfrac1n\sum(x_i-\bar x)^2$
\item 二项分布 $P(X=k)=\mathrm C_n^k p^k(1-p)^{n-k}$
\end{itemize}
\end{fastcard}

\begin{fastcard}{复数}
$z=a+bi$，$|z|=\sqrt{a^2+b^2}$，$i^2=-1$，$i$ 的幂周期为 4。
\end{fastcard}
