\begin{center}{\Large\bfseries 高一数学·必修一知识清单}\end{center}

\begin{multicols}{3}
\section{集合}
\begin{itemize}
\item 列举法、描述法、图示法
\item $A\cap B$，$A\cup B$，$\complement_U A$
\item 子集、真子集、集合相等
\end{itemize}
\section{常用逻辑用语}
\begin{itemize}
\item 充分条件、必要条件、充要条件
\item 全称量词与存在量词
\item 命题的否定与否命题
\end{itemize}
\section{基本不等式}
$a+b\ge 2\sqrt{ab}$（$a,b>0$），等号当且仅当 $a=b$。
\section{函数概念}
\begin{itemize}
\item 定义域、对应关系、值域
\item 单调性、奇偶性、周期性
\item 反函数与图象变换
\end{itemize}
\section{指数与对数}
\begin{itemize}
\item $a^m a^n=a^{m+n}$，$(a^m)^n=a^{mn}$
\item $\log_a(MN)=\log_a M+\log_a N$
\item 换底公式 $\log_a b=\dfrac{\ln b}{\ln a}$
\end{itemize}
\section{三角函数}
\begin{itemize}
\item $\sin^2x+\cos^2x=1$，$1+\tan^2x=\sec^2x$
\item 诱导公式：奇变偶不变，符号看象限
\item $y=A\sin(\omega x+\varphi)$ 的图象与性质
\end{itemize}
\section{平面向量}
\begin{itemize}
\item $\vec a\cdot\vec b=|\vec a||\vec b|\cos\theta$
\item 共线、垂直的坐标判定
\end{itemize}
\section{复数}
\begin{itemize}
\item $z=a+bi$，$|z|=\sqrt{a^2+b^2}$
\item $i^2=-1$，$i$ 的幂周期为 4
\end{itemize}
\section{数列}
\begin{itemize}
\item 等差：$a_n=a_1+(n-1)d$，$S_n=\dfrac{n(a_1+a_n)}{2}$
\item 等比：$a_n=a_1q^{n-1}$，$S_n=\dfrac{a_1(1-q^n)}{1-q}$
\item 递推：累加、累乘、构造法
\end{itemize}
\section{立体几何}
\begin{itemize}
\item 线面平行、垂直的判定与性质
\item 棱柱、棱锥、球的表面积与体积
\item 空间向量求角：$\cos\theta=\dfrac{|\vec a\cdot\vec b|}{|\vec a||\vec b|}$
\end{itemize}
\section{概率统计}
\begin{itemize}
\item 古典概型 $P(A)=\dfrac{m}{n}$
\item 均值 $\bar x=\dfrac1n\sum x_i$，方差 $s^2=\dfrac1n\sum(x_i-\bar x)^2$
\item 分层抽样与频率分布直方图
\end{itemize}
\section{导数与积分}
\begin{itemize}
\item $(x^n)'=nx^{n-1}$，$(\sin x)'=\cos x$
\item 单调性：$f'(x)>0\Rightarrow$ 递增
\item $\displaystyle\int x^n\,\mathrm{d}x=\dfrac{x^{n+1}}{n+1}+C$
\end{itemize}
