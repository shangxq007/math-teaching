\section*{高观点拓展：特征值是不动点吗？—— 从特征向量到莫比乌斯不动点}

\begin{remark}[核心结论]
\textbf{特征值本身不是不动点，特征向量才是“方向（一维子空间）”上的不动点}。当且仅当特征值 $\lambda = 1$ 时，特征向量才是绝对不动点；而在齐次坐标升维视角下，\textbf{矩阵特征向量的斜率（分量比值）就是分式递推的不动点，特征值之比则是等比构造的公比}。
\end{remark}

\subsection*{1. 几何辨析：绝对不动点 vs 方向不动点}
设线性变换 $T(\vec{v}) = A\vec{v}$：
\begin{itemize}
    \item \textbf{特征向量 $\vec{v}$ ($A\vec{v} = \lambda\vec{v}$)}：经过矩阵 $A$ 作用后，向量\textbf{所在直线的方向完全保持不变}，仅在长度上伸缩了 $\lambda$ 倍。若把“一条过原点的直线”视为射影空间中的一个点，则特征向量正是线性变换在方向上的\textbf{不动点}。
    \item \textbf{绝对不动点 ($A\vec{v} = \vec{v}$)}：当且仅当特征值 $\lambda = 1$ 时，向量方向与长度均不改变（如马尔可夫链的稳态分布向量）。
\end{itemize}

\subsection*{2. 惊人的同构：分式递推的“不动点”为何等于矩阵的“特征方向”？}
考虑分式递推（莫比乌斯变换） $x_{n+1} = \frac{a x_n + b}{c x_n + d}\ (ad - bc \neq 0)$：
\begin{enumerate}
    \item \textbf{升维（引入齐次坐标）}：将一维数轴上的数 $x_n$ 写成二维列向量的分量比值 $x_n = \frac{u_n}{v_n}$，则非线性分式递推瞬间化为纯粹的二维线性矩阵乘法：
    $$\begin{pmatrix} u_{n+1} \\ v_{n+1} \end{pmatrix} = \begin{pmatrix} a & b \\ c & d \end{pmatrix} \begin{pmatrix} u_n \\ v_n \end{pmatrix} = M \begin{pmatrix} u_n \\ v_n \end{pmatrix}$$
    \item \textbf{投影锁定不动点}：何时分式的值保持不变（$x_{n+1} = x_n = x^*$）？当且仅当变换后的向量与原向量\textbf{共线}（比值不变），这正是矩阵 $M$ 的特征向量定义：
    $$\begin{pmatrix} a & b \\ c & d \end{pmatrix} \begin{pmatrix} x^* \\ 1 \end{pmatrix} = \lambda \begin{pmatrix} x^* \\ 1 \end{pmatrix} \iff \begin{cases} ax^* + b = \lambda x^* \\ cx^* + d = \lambda \end{cases}$$
    消去 $\lambda$，立刻得到分式不动点方程 $x^* = \frac{ax^* + b}{cx^* + d}$，且特征值满足 $\lambda = cx^* + d$！
    \item \textbf{特征值之比即公比}：若矩阵 $M$ 有两个相异特征值 $\lambda_1 = cx_1 + d$ 与 $\lambda_2 = cx_2 + d$，代入我们之前推导的双不动点作比公式：
    $$\frac{x_{n+1} - x_1}{x_{n+1} - x_2} = \left(\frac{cx_2 + d}{cx_1 + d}\right) \frac{x_n - x_1}{x_n - x_2} = \left(\frac{\lambda_2}{\lambda_1}\right) \frac{x_n - x_1}{x_n - x_2}$$
    \textbf{结论}：双不动点构造等比数列的神秘公比，本质上就是转移矩阵两个特征值的比值！
\end{enumerate}

\begin{example}[以 $x_{n+1} = \frac{2x_n + 3}{x_n + 4}$ 验证同构]
对应矩阵 $M = \begin{pmatrix} 2 & 3 \\ 1 & 4 \end{pmatrix}$，求其特征方程 $\det(\lambda I - M) = (\lambda - 2)(\lambda - 4) - 3 = \lambda^2 - 6\lambda + 5 = 0$：
\begin{itemize}
    \item 特征值 $\lambda_1 = 5$ $\implies$ 特征向量 $\vec{v}_1 = \begin{pmatrix} 1 \\ 1 \end{pmatrix}$ $\implies$ 斜率比值 $x_1 = \frac{1}{1} = 1$（吸引不动点）；
    \item 特征值 $\lambda_2 = 1$ $\implies$ 特征向量 $\vec{v}_2 = \begin{pmatrix} -3 \\ 1 \end{pmatrix}$ $\implies$ 斜率比值 $x_2 = \frac{-3}{1} = -3$（排斥不动点）；
    \item 等比公比 $k = \frac{\lambda_2}{\lambda_1} = \frac{1}{5}$，与代数法完全一致！
\end{itemize}
\end{example}

二、 配套 Manim 动画完整脚本：特征向量直线投影为莫比乌斯不动点

这段动画（EigenvectorToFixedPoint）极其直观地演示了“二维线性空间 \(\rightarrow \) 一维投影线 \(y = 1\)”的射影降维魔法：
• 在二维平面上绘制矩阵 \(M = \begin{pmatrix} 2 & 3 \\ 1 & 4 \end{pmatrix}\) 的两条特征向量直线：\(u = v\)（对应 \(\lambda_1 = 5\)）和 \(u = -3v\)（对应 \(\lambda_2 = 1\)）；
• 施加矩阵线性变换 ApplyMatrix，观众亲眼看到普通向量发生偏转，而两条特征向量直线方向纹丝不动（只有向量沿直线伸缩）；
• 同时画出水平投影线 \(v = 1\)（即一维数轴 \(x = u/v\)），两条特征方向直线与 \(v = 1\) 的交点，恰好就是分式递推 \(x_{n+1} = \frac{2x_n+3}{x_n+4}\) 的两个不动点 \(x_1 = 1\) 和 \(x_2 = -3\)！
请保存为 eigen_to_fixed_point.py 运行：
python
from manim import *

class EigenvectorToFixedPoint(Scene):
    def construct(self):
        FONT_NAME = "SimHei"

        # 1. 顶部标题
        title = Text("射影同构：二维特征向量方向 = 一维分式递推不动点", font=FONT_NAME, font_size=24).to_edge(UP)
        self.play(Write(title))

        # 2. 建立二维齐次坐标系 (横轴 u, 纵轴 v)
        axes = Axes(
            x_range=[-4.5, 4.5, 1],
            y_range=[-1.5, 3.5, 1],
            x_length=9.5,
            y_length=5.2,
            axis_config={"include_tip": True, "font_size": 18}
        ).shift(DOWN * 0.4)
        labels = axes.get_axis_labels(x_label="u", y_label="v")

        # 画出投影水平线 v = 1 (一维数轴 x = u/v)
        proj_line = axes.plot(lambda u: 1, x_range=[-4.2, 4.2], color=YELLOW, stroke_width=3)
        proj_label = MathTex(r"\text{投影数轴 } v = 1 \ (x = \frac{u}{v})", font_size=22, color=YELLOW).next_to(axes.c2p(2.8, 1), UP, buff=0.15)

        self.play(Create(axes), Write(labels), Create(proj_line), Write(proj_label))

        # 3. 绘制矩阵 M = [[2, 3], [1, 4]] 的两条特征方向直线
        # 特征向量 1: (1, 1) -> 直线 v = u (对应 x_1 = 1, lambda_1 = 5)
        eigen_line1 = DashedLine(axes.c2p(-1.2, -1.2), axes.c2p(3.2, 3.2), color=GREEN, stroke_width=3)
        # 特征向量 2: (-3, 1) -> 直线 v = -u/3 (对应 x_2 = -3, lambda_2 = 1)
        eigen_line2 = DashedLine(axes.c2p(1.5, -0.5), axes.c2p(-4.2, 1.4), color=RED, stroke_width=3)

        vec1 = Arrow(axes.c2p(0, 0), axes.c2p(0.5, 0.5), buff=0, color=GREEN, stroke_width=5)
        vec2 = Arrow(axes.c2p(0, 0), axes.c2p(-1.8, 0.6), buff=0, color=RED, stroke_width=5)
        # 一个普通非特征向量 (0.5, 1.5)
        normal_vec = Arrow(axes.c2p(0, 0), axes.c2p(0.2, 0.8), buff=0, color=BLUE, stroke_width=5)

        self.play(
            Create(eigen_line1), Create(eigen_line2),
            GrowArrow(vec1), GrowArrow(vec2), GrowArrow(normal_vec)
        )
        self.wait(0.8)

        # 4. 施加矩阵变换演示：特征向量方向不变，普通向量发生偏转
        # 注意：为了画面不超出边界，将矩阵整体缩小 0.35 倍演示方向不变性 (0.35 * M)
        matrix_tex = MathTex(r"M = \begin{pmatrix} 2 & 3 \\ 1 & 4 \end{pmatrix}", font_size=26).to_corner(UL).shift(DOWN * 0.8 + RIGHT * 0.3)
        self.play(Write(matrix_tex))

        # 计算变换后的终点 (按比例 0.45 缩放以保持视觉清晰)
        # vec1 (0.5, 0.5) -> lambda_1=5 -> 沿原方向伸长
        vec1_after = Arrow(axes.c2p(0, 0), axes.c2p(1.8, 1.8), buff=0, color=GREEN, stroke_width=5)
        # vec2 (-1.8, 0.6) -> lambda_2=1 -> 长度方向全不变
        vec2_after = Arrow(axes.c2p(0, 0), axes.c2p(-1.8, 0.6), buff=0, color=RED, stroke_width=5)
        # normal_vec (0.2, 0.8) -> M*(0.2, 0.8) = (2.8, 3.4) -> 缩放到 (1.4, 1.7)，方向明显向右偏转
        normal_vec_after = Arrow(axes.c2p(0, 0), axes.c2p(1.4, 1.7), buff=0, color=BLUE, stroke_width=5)

        self.play(
            Transform(vec1, vec1_after),
            Transform(vec2, vec2_after),
            Transform(normal_vec, normal_vec_after),
            run_time=1.8
        )
        self.wait(0.8)

        # 5. 高亮特征直线与投影线 v = 1 的交点：正是双不动点 x_1 = 1 与 x_2 = -3
        fp1 = Dot(axes.c2p(1, 1), radius=0.1, color=GREEN)
        fp1_label = MathTex(r"x_1 = 1 \ (\lambda_1 = 5)", font_size=22, color=GREEN).next_to(fp1, DR, buff=0.15)

        fp2 = Dot(axes.c2p(-3, 1), radius=0.1, color=RED)
        fp2_label = MathTex(r"x_2 = -3 \ (\lambda_2 = 1)", font_size=22, color=RED).next_to(fp2, DL, buff=0.15)

        concl = MathTex(
            r"\text{特征方向落点} \implies x_{n+1} = \frac{2x_n+3}{x_n+4} \text{ 的双不动点，公比 } k = \frac{\lambda_2}{\lambda_1} = \frac{1}{5}",
            font_size=22, color=YELLOW
        ).to_edge(DOWN, buff=0.25)

        self.play(
            FadeIn(fp1), Write(fp1_label),
            FadeIn(fp2), Write(fp2_label),
            Write(concl)
        )
        self.wait(2.5)
