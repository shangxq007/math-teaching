\chapter{一元二次函数、方程和不等式}

\section{二次函数的解析式与图象}

\begin{problem}\level{易}
把 $y=2x^2-4x+1$ 配方成顶点式，并写出顶点坐标与对称轴。
\end{problem}
\begin{solution}
$y=2(x^2-2x)+1=2(x-1)^2-1$，顶点 $(1,-1)$，对称轴 $x=1$。
\end{solution}

\begin{problem}\level{易}
求 $y=x^2-4x+3$ 的顶点坐标、对称轴与最小值。
\end{problem}
\begin{solution}
$y=(x-2)^2-1$，顶点 $(2,-1)$，对称轴 $x=2$，最小值 $-1$。
\end{solution}

\begin{problem}\level{中}
求 $y=-x^2+2x+1$ 的顶点坐标、对称轴与最大值。
\end{problem}
\begin{solution}
$y=-(x-1)^2+2$，顶点 $(1,2)$，对称轴 $x=1$，最大值 $2$。
\end{solution}

\begin{problem}\level{中}
已知二次函数的图象过点 $(-1,0)$、$(3,0)$ 和 $(0,-3)$，求其解析式。
\end{problem}
\begin{solution}
设交点式 $y=a(x+1)(x-3)$，代入 $(0,-3)$ 得 $-3=a\cdot1\cdot(-3)$，故 $a=1$。
解析式为 $y=(x+1)(x-3)=x^2-2x-3$。
\end{solution}

\begin{problem}\level{难}
若二次函数 $y=x^2+bx+c$ 的图象顶点在 $x$ 轴上，且过点 $(1,4)$，求 $b$、$c$。
\end{problem}
\begin{solution}
顶点在 $x$ 轴上，故 $\Delta=b^2-4c=0$，即 $c=\dfrac{b^2}{4}$。
又过 $(1,4)$，得 $1+b+c=4$，即 $b+c=3$。
联立解得 $b=-2$，$c=4$（$b=6$ 时 $c=9$ 不满足 $b+c=3$，舍去）。
\end{solution}

\section{二次函数与方程、不等式}

\begin{problem}\level{易}
解方程 $x^2-5x+6=0$。
\end{problem}
\begin{solution}
$(x-2)(x-3)=0$，故 $x=2$ 或 $x=3$。
\end{solution}

\begin{problem}\level{易}
解不等式 $x^2-5x+6>0$。
\end{problem}
\begin{solution}
方程两根为 $2$、$3$，$a=1>0$，大于取两边，解集为 $(-\infty,2)\cup(3,+\infty)$。
\end{solution}

\begin{problem}\level{中}
解不等式 $x^2-4x+4\leq0$。
\end{problem}
\begin{solution}
$\Delta=0$，方程有重根 $x=2$。不等式化为 $(x-2)^2\leq0$，解集为 $\{2\}$。
\end{solution}

\begin{problem}\level{中}
解不等式 $x^2+2x+3>0$。
\end{problem}
\begin{solution}
$\Delta=4-12=-8<0$，方程无实根，又 $a=1>0$，故解集为 $\mathbb{R}$。
\end{solution}

\begin{problem}\level{中}
解不等式 $-x^2+2x+3>0$。
\end{problem}
\begin{solution}
两边同乘 $-1$ 得 $x^2-2x-3<0$，解集为 $(-1,3)$。
\end{solution}

\begin{problem}\level{难}
用数轴穿根法解不等式 $(x+1)(x-2)(x-4)<0$。
\end{problem}
\begin{solution}
根为 $-1$、$2$、$4$。从右上方起笔穿根，各段符号依次为 $+$、$-$、$+$、$-$。
取小于 $0$ 的部分，解集为 $(-\infty,-1)\cup(2,4)$。
\end{solution}

\begin{problem}\level{难}
解关于 $x$ 的不等式 $x^2-(a+1)x+a<0$（$a$ 为参数）。
\end{problem}
\begin{solution}
因式分解得 $(x-1)(x-a)<0$。
\begin{itemize}
  \item 若 $a>1$，解集为 $(1,a)$；
  \item 若 $a=1$，不等式为 $(x-1)^2<0$，解集为 $\emptyset$；
  \item 若 $a<1$，解集为 $(a,1)$。
\end{itemize}
\end{solution}

\begin{problem}\level{难}
解不等式 $\dfrac{x+1}{x-2}\geq0$。
\end{problem}
\begin{solution}
等价于 $(x+1)(x-2)\geq0$ 且 $x\neq2$。解集为 $(-\infty,-1]\cup(2,+\infty)$。
\end{solution}

\section{区间最值与恒成立}

\begin{problem}\level{易}
求 $y=x^2-2x-3$ 在区间 $[0,4]$ 上的最大值与最小值。
\end{problem}
\begin{solution}
$y=(x-1)^2-4$，对称轴 $x=1\in[0,4]$。最小值 $f(1)=-4$；
端点 $f(0)=-3$，$f(4)=5$，最大值为 $5$。
\end{solution}

\begin{problem}\level{中}
求 $y=-x^2+2x+1$ 在区间 $[0,3]$ 上的最大值与最小值。
\end{problem}
\begin{solution}
$y=-(x-1)^2+2$，对称轴 $x=1\in[0,3]$，$a<0$。
最大值 $f(1)=2$；端点 $f(0)=1$，$f(3)=-2$，最小值为 $-2$。
\end{solution}

\begin{problem}\level{中}
求 $y=x^2-2x-3$ 在区间 $[m,m+2]$ 上的最小值。
\end{problem}
\begin{solution}
对称轴 $x=1$。
\begin{itemize}
  \item 若 $1<m$，最小值为 $f(m)=m^2-2m-3$；
  \item 若 $m\leq1\leq m+2$，最小值为 $f(1)=-4$；
  \item 若 $1>m+2$，最小值为 $f(m+2)=m^2+2m-3$。
\end{itemize}
\end{solution}

\begin{problem}\level{中}
若 $x^2+2x+c>0$ 对任意实数 $x$ 恒成立，求 $c$ 的取值范围。
\end{problem}
\begin{solution}
只需 $\Delta=4-4c<0$，解得 $c>1$。
\end{solution}

\begin{problem}\level{中}
若 $x^2-2x+m\geq0$ 恒成立，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
只需 $\Delta=4-4m\leq0$，解得 $m\geq1$。
\end{solution}

\begin{problem}\level{中}
若 $x^2-2x+m>0$ 对任意 $x\in[0,3]$ 恒成立，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
设 $f(x)=x^2-2x+m$，对称轴 $x=1\in[0,3]$，
最小值 $f(1)=m-1>0$，得 $m>1$。
\end{solution}

\begin{problem}\level{难}
若 $x^2-2x+m>0$ 对任意 $x\in[3,4]$ 恒成立，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
对称轴 $x=1<3$，函数在 $[3,4]$ 上递增，最小值 $f(3)=m+3>0$，得 $m>-3$。
注意：若套 $\Delta<0$ 得 $m>1$，会漏掉 $(-3,1]$。
\end{solution}

\begin{problem}\level{难}
若存在 $x\in[0,2]$，使 $x^2-2x+m>0$，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
能成立看最大值。对称轴 $x=1\in[0,2]$，端点 $f(0)=f(2)=m$，
故最大值为 $m$，由 $m>0$ 得 $m>0$。
\end{solution}

\begin{problem}\level{难}
若 $-x^2+2x+m>0$ 对任意 $x\in[0,3]$ 恒成立，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
$a=-1<0$，开口向下，最小值在离对称轴 $x=1$ 较远的端点 $x=3$ 处，
$f(3)=m-3>0$，得 $m>3$。
\end{solution}

\begin{problem}\level{难}
若 $x^2-2x-3>0$ 对任意 $x\in[m,m+2]$ 恒成立，求 $m$ 的取值范围。
\end{problem}
\begin{solution}
对称轴 $x=1$。
\begin{itemize}
  \item 若 $1<m$，函数在 $[m,m+2]$ 上递增，最小值 $f(m)=m^2-2m-3>0$，得 $m>3$；
  \item 若 $m\leq1\leq m+2$，最小值 $f(1)=-4<0$，不可能；
  \item 若 $1>m+2$，函数在 $[m,m+2]$ 上递减，最小值 $f(m+2)=m^2+2m-3>0$，得 $m<-3$。
\end{itemize}
综上 $m>3$ 或 $m<-3$。
\end{solution}

\begin{problem}\level{难}
设 $p$：$x\in(-1,3)$，$q$：$x^2-2x-3<0$。判断 $p$ 是 $q$ 的什么条件。
\end{problem}
\begin{solution}
$q$ 的解集为 $(-1,3)$，与 $p$ 的集合相同，故 $p$ 是 $q$ 的充要条件。
\end{solution}

\begin{problem}\level{难}
设 $p$：$x>3$，$q$：$x^2-2x-3>0$。判断 $p$ 是 $q$ 的什么条件。
\end{problem}
\begin{solution}
$q$ 的解集为 $(-\infty,-1)\cup(3,+\infty)$，$p$ 的集合 $(3,+\infty)$ 是其真子集，
故 $p$ 是 $q$ 的充分不必要条件。
\end{solution}
