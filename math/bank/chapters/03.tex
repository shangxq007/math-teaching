\section{二次函数与方程、不等式}

\begin{problem}[易]
解方程 $x^2-5x+6=0$。
\end{problem}
\begin{solution}
$(x-2)(x-3)=0$，故 $x=2$ 或 $x=3$。
\end{solution}

\begin{problem}[易]
解不等式 $x^2-5x+6>0$。
\end{problem}
\begin{solution}
方程两根为 $2$、$3$，$a=1>0$，大于取两边，解集为 $(-\infty,2)\cup(3,+\infty)$。
\end{solution}

\begin{problem}[中]
解不等式 $x^2-4x+4\leq0$。
\end{problem}
\begin{solution}
$\Delta=0$，方程有重根 $x=2$。不等式化为 $(x-2)^2\leq0$，解集为 $\{2\}$。
\end{solution}

\begin{problem}[中]
解不等式 $x^2+2x+3>0$。
\end{problem}
\begin{solution}
$\Delta=4-12=-8<0$，方程无实根，又 $a=1>0$，故解集为 $\mathbb{R}$。
\end{solution}

\begin{problem}[中]
解不等式 $-x^2+2x+3>0$。
\end{problem}
\begin{solution}
两边同乘 $-1$ 得 $x^2-2x-3<0$，解集为 $(-1,3)$。
\end{solution}

\begin{problem}[难]
用数轴穿根法解不等式 $(x+1)(x-2)(x-4)<0$。
\end{problem}
\begin{solution}
根为 $-1$、$2$、$4$。从右上方起笔穿根，各段符号依次为 $+$、$-$、$+$、$-$。
取小于 $0$ 的部分，解集为 $(-\infty,-1)\cup(2,4)$。
\end{solution}

\begin{problem}[难]
解关于 $x$ 的不等式 $x^2-(a+1)x+a<0$（$a$ 为参数）。
\end{problem}
\begin{solution}
因式分解得 $(x-1)(x-a)<0$。
\begin{itemize}
  \item 若 $a>1$，解集为 $(1,a)$；
  \item 若 $a=1$，不等式为 $(x-1)^2<0$，解集为 $\emptyset$；
  \item 若 $a<1$，解集为 $(a,1)$。
\end{itemize}
\end{solution}
